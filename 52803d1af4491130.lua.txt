local function hover(btn,bright)
	btn.MouseEnter:Connect(function()
		TweenService:Create(btn,TweenInfo.new(0.18),{
			BackgroundColor3 = bright
		}):Play()
	end)

	btn.MouseLeave:Connect(function()
		TweenService:Create(btn,TweenInfo.new(0.18),{
			BackgroundColor3 = btn == beta and Color3.fromRGB(95,70,255) or Color3.fromRGB(45,45,55)
		}):Play()
	end)
end

hover(old,Color3.fromRGB(70,70,95))
hover(beta,Color3.fromRGB(125,95,255))

TweenService:Create(
	frame,
	TweenInfo.new(0.4,Enum.EasingStyle.Back),
	{Size = UDim2.new(0,400,0,240)}
):Play()

local function runScript(link)

	TweenService:Create(frame,TweenInfo.new(0.25),{
		Size = UDim2.new(0,0,0,0)
	}):Play()

	TweenService:Create(blur,TweenInfo.new(0.25),{
		Size = 0
	}):Play()

	task.wait(0.3)

	gui:Destroy()
	blur:Destroy()

	loadstring(game:HttpGet(link))()

end
