local function patchValidationGuards(functionName)
    for _, func in pairs(getgc(true)) do
        if typeof(func) == "function" and debug.getinfo(func).name == functionName then
            hookfunction(func, function(...)
                return true
            end)
        end
    end
end

local function getLocalCharacterModel()

local function getLocalRootPart()
    local character = getLocalCharacterModel()
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if humanoid and humanoid.RootPart then
        return humanoid.RootPart
    end
    return character and character:FindFirstChild("HumanoidRootPart")
end

local function isFriendTarget(player)
    if not player or player == LocalPlayer then
        return false
    end
    return LocalPlayer:IsFriendsWith(player.UserId)
end

local function isEligibleCombatTarget(character)
    local localCharacter = LocalPlayer.Character
    local rootPart = character and character:FindFirstChild("HumanoidRootPart")
    if not localCharacter or not character or not rootPart then
        return false
    end

    local localName = localCharacter.Name
    local cutscene = character:GetAttribute("Cutscene")
    local cutsceneValid = not cutscene or cutscene == localCharacter:GetAttribute("Cutscene")

    return not character:GetAttribute("Ignore")
        and cutsceneValid
        and (not character:GetAttribute("Grabbed") or character:GetAttribute("Grabbed") == localName)
        and (not character:GetAttribute("Victim") or character:GetAttribute("Victim") == localName)
        and not character:GetAttribute("Invincible")
        and not character:GetAttribute("Grabbing")
end

local function collectTargetsInRange()
    local root = getLocalRootPart()
    if not root then
        return {}
    end

    local targets = {}

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and (not Settings.InstantKillIgnoreFriends or not isFriendTarget(player)) and player:FindFirstChild(" ") == nil then
            local character = player.Character
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            local hrp = character and character:FindFirstChild("HumanoidRootPart")
            if humanoid and hrp and isEligibleCombatTarget(character) then
                local distance = (hrp.Position - root.Position).Magnitude
                if distance < Settings.RangeRadius then
                    table.insert(targets, {
                        character = character,
                        distance = distance,
                        health = humanoid:GetAttribute("Health") or humanoid.Health or 0,
                    })
                end
            end
        end
    end

    local mode = Settings.InstantKillTargetSelection or "All-in-range"
    if mode == "Lowest" then
        table.sort(targets, function(a, b)
            return a.health < b.health
        end)
        return targets[1] and { targets[1] } or {}
    end

    table.sort(targets, function(a, b)
        return a.distance < b.distance
    end)

    if mode == "Closest" then
        return targets[1] and { targets[1] } or {}
    end

    return targets
end

local function triggerDashRemote()
    local root = getLocalRootPart()
    if not root then
        return
    end

    require(ReplicatedStorage:WaitForChild("Core")).Library("Remote").Send("Dash", root.CFrame, "L", 1)
end

local function filterExcluded(bestTarget, validTargets)
    if not getgenv().isExcludedCharacter then
        return bestTarget, validTargets
    end

    if bestTarget and getgenv().isExcludedCharacter(bestTarget) then
        bestTarget = nil
    end

    if validTargets and type(validTargets) == "table" then
        for i = #validTargets, 1, -1 do
            if getgenv().isExcludedCharacter(validTargets[i]) then
                table.remove(validTargets, i)
            end
        end
    end

    return bestTarget, validTargets
end

local function triggerInstantKillBurst()
    local targets = collectTargetsInRange()
    if #targets == 0 then
        return
    end

    local data = LocalPlayer:FindFirstChild("Data")
    local characterValue = data and data:FindFirstChild("Character")
    local currentCharacter = characterValue and characterValue.Value
    if not currentCharacter or currentCharacter == "" then
        return
    end

    local characterFolder = ReplicatedStorage:FindFirstChild("Characters")
    local remotes = ReplicatedStorage:FindFirstChild("Remotes")
    if not characterFolder or not remotes then
        return
    end

    local charData = characterFolder:FindFirstChild(currentCharacter)
    local wallCombo = charData and charData:FindFirstChild("WallCombo")
    local abilityRemote = remotes:FindFirstChild("Abilities") and remotes.Abilities:FindFirstChild("Ability")
    local actionRemote = remotes:FindFirstChild("Combat") and remotes.Combat:FindFirstChild("Action")
    if not wallCombo or not abilityRemote or not actionRemote then
        return
    end

    local repeatCount = currentCharacter == "Gon" and 20 or 50
    local hitCharacters = {}
    local index = 1
    for _, target in ipairs(targets) do
        for _ = 1, repeatCount do
            hitCharacters[index] = target.character
            index += 1
        end
    end

    abilityRemote:FireServer(wallCombo, 69)
    actionRemote:FireServer(wallCombo, "", 4, 69, {
        BestHitCharacter = nil,
        HitCharacters = hitCharacters,
        Ignore = {},
        Actions = {},
    })
end

local function getCombatModule()

        local ok, core = pcall(function()

        if not ok or type(core) ~= "table" or type(core.Get) ~= "function" then

local function getHitboxSize(originalArg)
    if Settings.HitboxMethod == "Add" then
        local originalSize = typeof(originalArg.Size) == "Vector3" and originalArg.Size or Vector3.new(0, 0, 0)
        return Vector3.new(
            originalSize.X + Settings.HitboxX,
            originalSize.Y + Settings.HitboxY,
            originalSize.Z + Settings.HitboxZ
        )
    end

    return Vector3.new(Settings.HitboxX, Settings.HitboxY, Settings.HitboxZ)
end

local function visualizeHitbox(args, newSize)
    if not Settings.HitboxVisualizer then
        return
    end

    local cf = args[3].CFrame or args[3].cf or args[3].Frame
    if not cf and typeof(args[2]) == "CFrame" then
        cf = args[2]
    elseif not cf and typeof(args[2]) == "Instance" and args[2]:IsA("BasePart") then
        cf = args[2].CFrame
    end
    if not cf then
        local root = getLocalRootPart()
        cf = root and root.CFrame or CFrame.new()
    end

    local part = Instance.new("Part")
    part.Name = "HitboxVisualizer"
    part.Anchored = true
    part.CanCollide = false
    part.Material = Enum.Material.ForceField
    part.Color = Color3.fromRGB(255, 0, 0)
    part.Transparency = 0.7
    part.Size = newSize
    part.CFrame = cf
    part.CastShadow = false
    part.Parent = workspace
    Debris:AddItem(part, 0.08)
end

local function applyHitboxHook()
    local combat = getCombatModule()
    if not combat or type(combat.Box) ~= "function" then
        return
    end

    if not hitboxState.originalHitBox then
        hitboxState.originalHitBox = combat.Box
    end

    if not Settings.HitboxEnabled then
        combat.Box = hitboxState.originalHitBox
        return
    end

    combat.Box = function(...)
        local args = { ... }
        if not args[3] or type(args[3]) ~= "table" then
            return hitboxState.originalHitBox(...)
        end

        local newSize = getHitboxSize(args[3])
        visualizeHitbox(args, newSize)

        if Settings.HitboxLegitify then
            local modifiedArg3 = {}
            for key, value in pairs(args[3]) do
                modifiedArg3[key] = value
            end
            modifiedArg3.Size = newSize
            local bestTarget, validTargets = hitboxState.originalHitBox(args[1], args[2], modifiedArg3)
            return filterExcluded(bestTarget, validTargets)
        end

        local bestTarget, validTargets = hitboxState.originalHitBox(args[1], args[2], { Size = newSize })
        return filterExcluded(bestTarget, validTargets)
    end
end

local function applyEnhancerHook()
    local hit = getHitModule()
    if not hit or type(hit.Process) ~= "function" then
        return
    end

    if not hitboxState.originalHitProcess then
        hitboxState.originalHitProcess = hit.Process
    end

    if not Settings.EnhancerEnabled then
        hit.Process = hitboxState.originalHitProcess
        return
    end

    hit.Process = function(...)
        local bestTarget, validTargets, blockedTargets = hitboxState.originalHitProcess(...)
        bestTarget, validTargets = filterExcluded(bestTarget, validTargets)

        if not validTargets or #validTargets == 0 then
            return bestTarget, validTargets, blockedTargets
        end

        if not Settings.HitboxLegitify then
            triggerDashRemote()
        end

        pcall(function()
            local dataFolder = LocalPlayer:WaitForChild("Data", 5)
            if not dataFolder then
                return
            end

            local characterValue = dataFolder:WaitForChild("Character", 5)
            if not characterValue then
                return
            end

            local charValue = characterValue.Value
            if not charValue then
                return
            end

            local characterData = ReplicatedStorage:WaitForChild("Characters", 5):FindFirstChild(charValue)
            if not characterData then
                return
            end

            local abilityRemote = ReplicatedStorage.Remotes.Abilities.Ability
            local actionRemote = ReplicatedStorage.Remotes.Combat.Action

            if Settings.EnhancerMultiplier == 0 then
                abilityRemote:FireServer(nil, 69)
                actionRemote:FireServer(nil, "", 4, 69, {
                    BestHitCharacter = nil,
                    HitCharacters = validTargets,
                    Ignore = {},
                    Actions = {},
                })
                return
            end

            local wallCombo = characterData:FindFirstChild("WallCombo")
            if wallCombo then
                for _ = 1, Settings.EnhancerMultiplier do
                    abilityRemote:FireServer(wallCombo, 69)
                    actionRemote:FireServer(wallCombo, "", 4, 69, {
                        BestHitCharacter = nil,
                        HitCharacters = validTargets,
                        Ignore = {},
                        Actions = {},
                    })
                end
            end
        end)

        return bestTarget, validTargets, blockedTargets
    end
end

local function normalizeColorInput(value, g, b, fallback)
    if typeof(value) == "Color3" then
        return value
    end

    if type(value) == "number" and type(g) == "number" and type(b) == "number" then
        if value <= 1 and g <= 1 and b <= 1 then
            return Color3.new(value, g, b)
        end
        return Color3.fromRGB(value, g, b)
    end

    if type(value) == "table" then
        local r = value.R or value.r or value[1]
        local g2 = value.G or value.g or value[2]
        local b2 = value.B or value.b or value[3]
        if type(r) == "number" and type(g2) == "number" and type(b2) == "number" then
            if r <= 1 and g2 <= 1 and b2 <= 1 then
                return Color3.new(r, g2, b2)
            end
            return Color3.fromRGB(r, g2, b2)
        end
    end

    return fallback
end

                        exploitState.savedPosition = currentRootPart.CFrame
                        instantReset()
                    end
                end)
            else
                exploitState.savedPosition = rootPart.CFrame
                instantReset()
            end
        elseif string.sub(healthStr, 1, 1) ~= "0" then
            exploitState.hasTriggered = false
        end
    end)

    exploitState.respawnHandler = LocalPlayer.CharacterAdded:Connect(function(char)
        if not Settings.InstantRespawn or not exploitState.savedPosition then
            return
        end

        local rootPart = char:WaitForChild("HumanoidRootPart", 5)
        if not rootPart then
            return
        end

        task.wait(0.2)
        pcall(function()
            require(LocalPlayer.PlayerScripts.Character.FullCustomReplication).Override(char, exploitState.savedPosition)
        end)
        exploitState.savedPosition = nil
    end)
end

    end

local function getBlockRemote()

local function fireBlockState(state)
    local remote = getBlockRemote()
    if not remote then
        return false
    end
    return pcall(function()
        remote:FireServer(state)
    end)
end

local function isAbilityAnimation(track)
    local animation = track and track.Animation
    local animationId = animation and animation.AnimationId
    return animationId and legitAbilityAnimations[animationId] or false
end

local function getThreatAnimations()
    local myCharacter = LocalPlayer.Character
    local myRoot = myCharacter and myCharacter:FindFirstChild("HumanoidRootPart")
    if not myRoot then
        return {}
    end

    local threats = {}

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local character = player.Character
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if humanoid and root and isEligibleCombatTarget(character) then
                local distance = (myRoot.Position - root.Position).Magnitude
                if distance <= Settings.LegitAutoBlockRange then
                    local foundTrack
                    for _, track in ipairs(humanoid:GetPlayingAnimationTracks()) do
                        if track.IsPlaying and isAbilityAnimation(track) then
                            foundTrack = track
                            break
                        end
                    end

                    table.insert(threats, {
                        root = root,
                        track = foundTrack,
                        distance = distance,
                    })
                end
            end
        end
    end

    table.sort(threats, function(a, b)
        return a.distance < b.distance
    end)

    return threats
end

local function getLocalBlockMode()
    local character = LocalPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not character or not humanoid then
        return false
    end

    for _, track in ipairs(humanoid:GetPlayingAnimationTracks()) do
        local animation = track.Animation
        local animationId = animation and animation.AnimationId
        if track.IsPlaying and animationId and legitNoTurnAnimations[animationId] then
            return "noturn", track
        end
    end

    local abilityValue = character:GetAttribute("Ability")
    if type(abilityValue) == "string" and legitNoTurnAbilities[abilityValue] then
        return "noturn", abilityValue
    end

    for _, track in ipairs(humanoid:GetPlayingAnimationTracks()) do
        if track.IsPlaying and isAbilityAnimation(track) then
            return true, track
        end
    end

    return false
end

local function disableLegitAutoBlock()
    local state = exploitState.legitAutoBlock

    if state.renderConnection then
        state.renderConnection:Disconnect()
        state.renderConnection = nil
    end

    if state.bound then
        pcall(function()
            RunService:UnbindFromRenderStep("KZLegitAutoBlock")
        end)
        state.bound = false
    end

    local character = LocalPlayer.Character
    fireBlockState(false)
    if character then
        pcall(function()
            character:SetAttribute("LocalBlock", false)
        end)
    end

    state.blockTimer = 0
    state.lastBlockWanted = false
    state.animBlockCache = {}
end

local function updateLegitAutoBlock()
    if not Settings.LegitAutoBlock then
        return
    end

    local state = exploitState.legitAutoBlock
    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if not character or not root then
        return
    end

    if character:GetAttribute("Ability") == nil then
        pcall(function()
            character:SetAttribute("Ability", "")
        end)
    end

    local blockMode, currentAnim = getLocalBlockMode()
    if blockMode == "noturn" and currentAnim then
        fireBlockState(false)
        pcall(function()
            character:SetAttribute("LocalBlock", false)
        end)
        state.blockTimer = tick() + ((type(currentAnim) == "userdata" and currentAnim.Length) or 0.7)
        state.lastBlockWanted = false
        state.animBlockCache = {}
        return
    end

    if blockMode == true then
        fireBlockState(false)
        pcall(function()
            character:SetAttribute("LocalBlock", false)
        end)
        state.blockTimer = 0
        state.lastBlockWanted = false
        state.animBlockCache = {}
        return
    end

    local threats = getThreatAnimations()
    local blockAny = false
    local abilityValue = character:GetAttribute("Ability")
    local canTurn = not (type(abilityValue) == "string" and legitNoTurnAbilities[abilityValue])

    for _, threat in ipairs(threats) do
        if threat.root and canTurn then
            local direction = threat.root.Position - root.Position
            direction = Vector3.new(direction.X, 0, direction.Z)
            if direction.Magnitude > 0.001 then
                root.CFrame = CFrame.lookAt(root.Position, root.Position + direction.Unit)
            end
        end

        if threat.track then
            if not state.animBlockCache[threat.track] then
                state.animBlockCache[threat.track] = true
                state.blockTimer = math.max(state.blockTimer or 0, tick() + (threat.track.Length or 0.2) + Settings.LegitAutoBlockExtend)
            end
            if tick() < (state.blockTimer or 0) then
                fireBlockState(true)
                pcall(function()
                    character:SetAttribute("LocalBlock", true)
                end)
            end
        else
            state.blockTimer = math.max(state.blockTimer or 0, tick() + 0.15)
            fireBlockState(true)
            pcall(function()
                character:SetAttribute("LocalBlock", true)
            end)
        end

        blockAny = true
    end

    if not blockAny and tick() > (state.blockTimer or 0) then
        fireBlockState(false)
        pcall(function()
            character:SetAttribute("LocalBlock", false)
        end)
        state.lastBlockWanted = false
        state.animBlockCache = {}
        return
    end

    state.lastBlockWanted = blockAny
end

local function enableLegitAutoBlock()
    disableLegitAutoBlock()

    local state = exploitState.legitAutoBlock
    state.renderConnection = RunService.RenderStepped:Connect(function()
        if not Settings.LegitAutoBlock then
            return
        end
        if not state.lastBlockWanted or tick() >= (state.blockTimer or 0) then
            return
        end

        local character = LocalPlayer.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if not character or not root then
            return
        end

        local abilityValue = character:GetAttribute("Ability")
        local canTurn = not (type(abilityValue) == "string" and legitNoTurnAbilities[abilityValue])
        if canTurn then
            for _, threat in ipairs(getThreatAnimations()) do
                if threat.root then
                    local direction = threat.root.Position - root.Position
                    direction = Vector3.new(direction.X, 0, direction.Z)
                    if direction.Magnitude > 0.001 then
                        root.CFrame = CFrame.lookAt(root.Position, root.Position + direction.Unit)
                    end
                end
            end
        end

        if character:GetAttribute("LocalBlock") ~= true then
            fireBlockState(true)
            pcall(function()
                character:SetAttribute("LocalBlock", true)
            end)
        end
    end)

    pcall(function()
        RunService:BindToRenderStep("KZLegitAutoBlock", Enum.RenderPriority.Camera.Value + 1, function()
            if not Settings.LegitAutoBlock then
                fireBlockState(false)
                local character = LocalPlayer.Character
                if character then
                    pcall(function()
                        character:SetAttribute("LocalBlock", false)
                    end)
                end
                state.lastBlockWanted = false
                state.blockTimer = 0
                state.animBlockCache = {}
                return
            end

            updateLegitAutoBlock()
        end)
        state.bound = true
    end)
end

local function scanDashFunction(functionName)
    local targetScript = LocalPlayer:WaitForChild("PlayerScripts"):WaitForChild("Combat"):WaitForChild("Dash")
    for _, func in pairs(getgc(true)) do
        if typeof(func) == "function" then
            local environment = getfenv(func)
            local scriptRef = environment and environment.script
            if scriptRef == targetScript and debug.getinfo(func).name == functionName then
                return func
            end
        end
    end
    return nil
end

local function dashRunAttack(p105, p106, p107, p108)
    local core = require(ReplicatedStorage:WaitForChild("Core"))
    local cameraCFrame = core.Services.Camera.CFrame
    local startedAt = os.clock()
    local actionId = tostring(startedAt)
    local yawTotal = 0
    local maxYaw = 0
    local minYaw = 0
    local elapsed = 0
    local hitCharacters, bestHit, guaranteeCharacters, guaranteeBest, verticalPass
    local upperHit, cachedUnknown
    local attackBlocked, attackRunning = false, false

    local function verticalValidation(character, heightOffset, tolerance, requireFalling, requireAbove)
        local ragdollFrame = core.Get("Combat", "Ragdoll").GetRagdollFrame(character)
        local selfCFrame = core.Get("Character", "FullCustomReplication").GetCFrame(p105)
        local targetCFrame = core.Get("Character", "FullCustomReplication").GetCFrame(character)
        local validHeight = not tolerance
        if not validHeight then
            if selfCFrame and targetCFrame then
                local diff = heightOffset + selfCFrame.Position.Y - targetCFrame.Position.Y
                validHeight = math.abs(diff) < tolerance
            end
        end
        local validFalling = not requireFalling
        if validFalling then
            ragdollFrame = validFalling
        elseif ragdollFrame then
            ragdollFrame = ragdollFrame.Velocity.Y < 0
        end
        local validAbove = not requireAbove
        if validAbove then
            targetCFrame = validAbove
        elseif selfCFrame and targetCFrame then
            targetCFrame = targetCFrame.Position.Y > selfCFrame.Position.Y
        else
            targetCFrame = selfCFrame
        end
        if validHeight then
            if not ragdollFrame then
                targetCFrame = ragdollFrame
            end
        else
            targetCFrame = validHeight
        end
        verticalPass = targetCFrame
        return verticalPass
    end

    local function upperValidation(character, part)
        local ragdollFrame = core.Get("Combat", "Ragdoll").GetRagdollFrame(character)
        local upVelocity = core.Get("Combat", "Ragdoll").UpVelocities[character]
        if ragdollFrame then
            if ragdollFrame.Velocity.Y < 0 and upVelocity then
                upVelocity = upVelocity > 1 and part.CollisionGroup ~= "NoCharacterCollisions"
            else
                upVelocity = false
            end
        end
        return upVelocity
    end

    local function runHitbox(size, offset, customValidation)
        if core.Library("Instance").Exists(p106) then
            local best1, list1 = core.Get("Combat", "Hit").Box(nil, p105, {
                Size = size,
                Offset = offset,
                IgnoreJump = true,
                IgnoreRagdolls = "Ground",
                IgnoreKnockback = true,
                NoBaseValidation = true,
                RequireInFront = true,
                BlockHitsThroughWalls = true,
                CustomValidation = customValidation
            })
            bestHit, hitCharacters = best1, list1
            local best2, list2 = core.Get("Combat", "Hit").Box(nil, p105, {
                Size = size,
                Offset = offset,
                IgnoreJump = true,
                IgnoreRagdolls = "Ground",
                IgnoreKnockback = true,
                RequireInFront = true,
                BlockHitsThroughWalls = true,
                CustomValidation = customValidation
            })
            guaranteeBest, guaranteeCharacters = best2, list2
            return #hitCharacters > 0
        end
    end

    local cachedGuarantee, cachedMinYaw, cachedMaxYaw, cachedUpper = guaranteeBest, maxYaw, minYaw, verticalPass
    local cachedUpperHit = nil

    local function groundedValidation(character, part)
        local valid = not character:GetAttribute("Ragdoll") or verticalValidation(character, 2, 0.25, true)
        if valid then
            valid = part.CollisionGroup ~= "NoCharacterCollisions"
        end
        return valid
    end

    while true do
        task.wait()
        local currentCamera = core.Services.Camera.CFrame
        local _, oldYaw, _ = cameraCFrame:ToOrientation()
        local _, newYaw, _ = currentCamera:ToOrientation()
        local yawDiff = math.deg(oldYaw - newYaw)
        if yawDiff > 180 then
            yawDiff = yawDiff - 360
        elseif yawDiff < -180 then
            yawDiff = yawDiff + 360
        end

        yawTotal += yawDiff
        maxYaw = math.max(cachedMaxYaw, yawTotal)
        minYaw = math.min(cachedMinYaw, yawTotal)
        elapsed = os.clock() - startedAt
        local remaining = p107 - elapsed

        if elapsed > 0.2125 then
            upperHit = runHitbox(Vector3.new(1, 1, 1), CFrame.new(0, 7.5, -0.5), upperValidation)
            cachedUpper = upperHit or runHitbox(Vector3.new(5, 5, 4.5), CFrame.new(0, 0, -2), groundedValidation)
            cachedUpperHit = upperHit
        end

        if cachedUpper or p108 < elapsed then
            local useFromCFrame = remaining > 0 and true or cachedUnknown
            local success = cachedUpper or runHitbox(Vector3.new(7.5, 5, 7), CFrame.new(0, 0, -2.75), groundedValidation)
            if success then
                core.Get("Character", "Move").SetJumpOverride("DashPunch", 0)
                task.delay(0.5, core.Get("Character", "Move").SetJumpOverride, "DashPunch", nil)
            end

            local function executeAttack()
                local actionData = {
                    Guarantees = cachedGuarantee,
                    Replace = function(target)
                        local threshold = core.Get("Combat", "Knockback").CharacterPresets[target] == "Uppercut" and 0.375 or 0.2625
                        local ok1, ragdoll = pcall(function() return core.Get("Combat", "Ragdoll").GetRagdoll(target) end)
                        if not ok1 then ragdoll = nil end
                        local ok2, ragdollFrame = pcall(function() return core.Get("Combat", "Ragdoll").GetRagdollFrame(target) end)
                        if not ok2 then ragdollFrame = nil end
                        local attrRagdoll = false
                        if target and target.GetAttribute then
                            pcall(function() attrRagdoll = target:GetAttribute("Ragdoll") end)
                        end
                        local isRagdolled = (ragdoll ~= nil) or (ragdollFrame ~= nil) or (attrRagdoll == true)
                        local forceDirection = nil
                        local ok3, targetCFrame = pcall(function() return target and core.Get("Character", "FullCustomReplication").GetCFrame(target) end)
                        if ok3 and targetCFrame then
                            local point = p106.CFrame:PointToObjectSpace(targetCFrame.Position)
                            local angle = point.X * 180 / 2
                            forceDirection = p106.CFrame * CFrame.Angles(0, math.rad(-math.clamp(angle, -180, 180)), 0)
                        end
                        local hasKnockback = false
                        if target and target.GetAttribute then
                            local okKB, kb = pcall(function() return target:GetAttribute("Knockback") end)
                            if okKB and kb then hasKnockback = true end
                        end
                        if isRagdolled then
                            return { ActionNumbers = { 2, 6 }, ForceDirection = forceDirection }
                        end
                        if elapsed < threshold then
                            if target and target.SetAttribute then
                                pcall(function() target:SetAttribute("Knockback", false) end)
                            end
                            return { ActionNumbers = { 1, 6 }, ForceDirection = nil }
                        end
                        local useThird
                        if maxYaw - minYaw > 180 then
                            useThird = not ragdoll or verticalValidation(target, 0, 2, nil, true)
                        else
                            useThird = false
                        end
                        local useAction3 = false
                        local useAction4 = useFromCFrame
                        if useAction4 then useAction4 = not ragdoll end
                        if useAction3 then
                            if ragdoll then ragdoll = cachedUpperHit end
                        else
                            ragdoll = useAction3
                        end
                        if useAction3 then useAction3 = not ragdoll end
                        if elapsed >= threshold then
                            pcall(function()
                                if target and target.SetAttribute then
                                    target:SetAttribute("Knockback", false)
                                end
                            end)
                        end
                        local pick = 1
                        if hasKnockback then pick = 5
                        elseif useAction4 then pick = 4
                        elseif useAction3 then pick = 3
                        elseif ragdoll then pick = 2 end
                        return { ActionNumbers = { pick, 6 }, ForceDirection = useAction4 and forceDirection or nil }
                    end,
                    UseFromCFrame = useFromCFrame and p106.CFrame * CFrame.new(0, 0, -2) or nil
                }
                task.wait(0.15)
                attackRunning = true
                local result, extra, _, _, hitData = core.Get("Combat", "Action").Event(nil, p105, "PunchDash", actionId, actionData, 0)
                if result then
                    if exploitState.legitDash.vfx then
                        exploitState.legitDash.vfx("DashHit", result, p106.Position)
                    end
                    core.Get("Camera", "Shake").Shake(nil, nil, {
                        Amplitude = 1.25,
                        Frequency = 0.0875,
                        FadeInTime = 0.025,
                        FadeOutTime = 0.5
                    }, nil, p105, table.unpack(extra))
                end
                local blocked = false
                for _, blockedCharacter in ipairs(hitData and (hitData.BlockedCharacters or {}) or {}) do
                    if not table.find(hitData.HitCharacters, blockedCharacter) then
                        blocked = true
                    end
                end
                attackBlocked = blocked
                attackRunning = false
            end

            task.spawn(executeAttack)
            if remaining > 0 then
                core.Services.Run.Heartbeat:Wait()
                task.wait(remaining + 0.025)
            end
            if exploitState.legitDash.punchAnimation then
                task.spawn(exploitState.legitDash.punchAnimation, p105, actionId)
            end
            return remaining > 0, success
        end

        cachedMinYaw, cachedMaxYaw, cameraCFrame = minYaw, maxYaw, currentCamera
    end
end

    exploitState.legitDash.targetRunAttackFunc = nil
    exploitState.legitDash.punchAnimation = nil
    exploitState.legitDash.vfx = nil

local function getCooldownMultiplier()
    local settingsFolder = ReplicatedStorage:FindFirstChild("Settings")
    local cooldowns = settingsFolder and settingsFolder:FindFirstChild("Cooldowns")
    local cooldownValue = cooldowns and (cooldowns:FindFirstChild("Evasive") or cooldowns:FindFirstChild("Ragdoll"))
    return cooldownValue and (cooldownValue.Value / 100) or 1
end

local function getRagdollColor(progress)
    if progress > 0.5 then
        return legitRagdollEspConfig.ColorMid:Lerp(legitRagdollEspConfig.ColorHigh, (progress - 0.5) * 2)
    end
    return legitRagdollEspConfig.ColorLow:Lerp(legitRagdollEspConfig.ColorMid, progress * 2)
end

local function createLegitRagdollEsp(player)
    if player == LocalPlayer or exploitState.legitEsp.ragdollData[player] or not Drawing then
        return
    end

    local text = Drawing.new("Text")
    text.Center = true
    text.Size = legitRagdollEspConfig.TextSize
    text.Outline = legitRagdollEspConfig.TextOutline
    text.OutlineColor = legitRagdollEspConfig.OutlineColor
    text.Font = legitRagdollEspConfig.TextFont
    text.Visible = false

    exploitState.legitEsp.ragdollData[player] = { Text = text }
end

local function removeLegitRagdollEsp(player)
    local data = exploitState.legitEsp.ragdollData[player]
    if data then
        data.Text:Remove()
        exploitState.legitEsp.ragdollData[player] = nil
    end
end

local function stopLegitRagdollEsp()
    local state = exploitState.legitEsp
    if state.ragdollRenderConnection then
        state.ragdollRenderConnection:Disconnect()
        state.ragdollRenderConnection = nil
    end
    if state.ragdollPlayerAddedConnection then
        state.ragdollPlayerAddedConnection:Disconnect()
        state.ragdollPlayerAddedConnection = nil
    end
    if state.ragdollPlayerRemovingConnection then
        state.ragdollPlayerRemovingConnection:Disconnect()
        state.ragdollPlayerRemovingConnection = nil
    end
    for player in pairs(state.ragdollData) do
        removeLegitRagdollEsp(player)
    end
end

local function startLegitRagdollEsp()
    stopLegitRagdollEsp()
    local ragdollModule = ensureRagdollModule()
    if not ragdollModule or not Drawing then
        return
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            createLegitRagdollEsp(player)
        end
    end

    exploitState.legitEsp.ragdollPlayerAddedConnection = Players.PlayerAdded:Connect(createLegitRagdollEsp)
    exploitState.legitEsp.ragdollPlayerRemovingConnection = Players.PlayerRemoving:Connect(removeLegitRagdollEsp)
    exploitState.legitEsp.ragdollRenderConnection = RunService.RenderStepped:Connect(function()
        for player, data in pairs(exploitState.legitEsp.ragdollData) do
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not root then
                data.Text.Visible = false
                continue
            end

            local ragdollStart = character:GetAttribute("Ragdoll")
            local endClock = ragdollModule.EndClocks and ragdollModule.EndClocks[character]
            local startClock = ragdollModule.StartClocks and ragdollModule.StartClocks[character]
            if typeof(ragdollStart) == "number" and endClock and startClock then
                local remaining = math.max(endClock - os.clock(), 0)
                local total = endClock - startClock
                local progress = total > 0 and (remaining / total) or 0
                local screenPos, onScreen = workspace.CurrentCamera:WorldToViewportPoint(root.Position + Vector3.new(0, legitRagdollEspConfig.OffsetY, 0))
                if onScreen and remaining > 0 then
                    data.Text.Text = string.format("%.1fs", remaining)
                    data.Text.Color = getRagdollColor(progress)
                    data.Text.Position = Vector2.new(screenPos.X, screenPos.Y)
                    data.Text.Visible = true
                else
                    data.Text.Visible = false
                end
            else
                data.Text.Visible = false
            end
        end
    end)
end

        duration = legitEvasiveEspConfig.BaseDuration * getCooldownMultiplier(),
    }
end

local function getEvasiveRemaining(player)
    local data = exploitState.legitEsp.evasiveCooldowns[player]
    if not data then
        return 0
    end

    local remaining = data.duration - (os.clock() - data.start)
    if remaining <= 0 then
        exploitState.legitEsp.evasiveCooldowns[player] = nil
        return 0
    end
    return remaining
end

local function monitorEvasivePlayer(player)
    exploitState.legitEsp.evasiveStates[player] = {
        wasRagdoll = false,
        wasDash = false,
    }

    local function onCharacter(character)
        local function update()
            local state = exploitState.legitEsp.evasiveStates[player]
            if not state then
                return
            end

            local ragdoll = character:GetAttribute("Ragdoll")
            local dash = character:GetAttribute("Dash")
            if state.wasRagdoll and dash and not state.wasDash then
                startEvasiveCooldown(player)
            end

            state.wasRagdoll = ragdoll
            state.wasDash = dash
        end

        character:GetAttributeChangedSignal("Ragdoll"):Connect(update)
        character:GetAttributeChangedSignal("Dash"):Connect(update)
        update()
    end

    if player.Character then
        onCharacter(player.Character)
    end
    player.CharacterAdded:Connect(onCharacter)
end

local function createLegitEvasiveEsp(player)
    if exploitState.legitEsp.evasiveData[player] or not Drawing then
        return
    end

    local text = Drawing.new("Text")
    text.Center = true
    text.Size = legitEvasiveEspConfig.TextSize
    text.Font = legitEvasiveEspConfig.Font
    text.Outline = legitEvasiveEspConfig.Outline
    text.OutlineColor = legitEvasiveEspConfig.OutlineColor
    text.Visible = false

    exploitState.legitEsp.evasiveData[player] = { Text = text }
end

local function removeLegitEvasiveEsp(player)
    local data = exploitState.legitEsp.evasiveData[player]
    if data then
        data.Text:Remove()
        exploitState.legitEsp.evasiveData[player] = nil
    end
    exploitState.legitEsp.evasiveCooldowns[player] = nil
    exploitState.legitEsp.evasiveStates[player] = nil
end

local function stopLegitEvasiveEsp()
    local state = exploitState.legitEsp
    if state.evasiveRenderConnection then
        state.evasiveRenderConnection:Disconnect()
        state.evasiveRenderConnection = nil
    end
    if state.evasivePlayerAddedConnection then
        state.evasivePlayerAddedConnection:Disconnect()
        state.evasivePlayerAddedConnection = nil
    end
    if state.evasivePlayerRemovingConnection then
        state.evasivePlayerRemovingConnection:Disconnect()
        state.evasivePlayerRemovingConnection = nil
    end
    for player in pairs(state.evasiveData) do
        removeLegitEvasiveEsp(player)
    end
end

local function startLegitEvasiveEsp()
    stopLegitEvasiveEsp()
    if not Drawing then
        return
    end

    for _, player in ipairs(Players:GetPlayers()) do
        monitorEvasivePlayer(player)
        createLegitEvasiveEsp(player)
    end

    exploitState.legitEsp.evasivePlayerAddedConnection = Players.PlayerAdded:Connect(function(player)
        monitorEvasivePlayer(player)
        createLegitEvasiveEsp(player)
    end)
    exploitState.legitEsp.evasivePlayerRemovingConnection = Players.PlayerRemoving:Connect(removeLegitEvasiveEsp)
    exploitState.legitEsp.evasiveRenderConnection = RunService.RenderStepped:Connect(function()
        for player, data in pairs(exploitState.legitEsp.evasiveData) do
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not root then
                data.Text.Visible = false
                continue
            end

            local remaining = getEvasiveRemaining(player)
            if player == LocalPlayer then
                data.Text.Text = remaining > 0 and string.format("Evasive: %.1fs", remaining) or "Evasive: READY"
                data.Text.Color = remaining > 0 and legitEvasiveEspConfig.ColorCooldown or legitEvasiveEspConfig.ColorReady
                data.Text.Position = Vector2.new(100, 100)
                data.Text.Visible = true
            else
                local screenPos, onScreen = workspace.CurrentCamera:WorldToViewportPoint(root.Position + Vector3.new(0, legitEvasiveEspConfig.OffsetY, 0))
                if not onScreen then
                    data.Text.Visible = false
                    continue
                end

                data.Text.Text = remaining > 0 and string.format("%.1fs", remaining) or "EVASIVE: READY"
                data.Text.Color = remaining > 0 and legitEvasiveEspConfig.ColorCooldown or legitEvasiveEspConfig.ColorReady
                data.Text.Position = Vector2.new(screenPos.X, screenPos.Y)
                data.Text.Visible = true
            end
        end
    end)
end

local function createInvisibleMirrorClone()
    local character = LocalPlayer.Character
    if not character then
        return nil
    end

    character.Archivable = true
    local clone = character:Clone()
    clone.Name = "MirrorClone"
    clone.Parent = workspace

    for _, d in ipairs(clone:GetDescendants()) do
        if d:IsA("Script") or d:IsA("LocalScript") then
            d:Destroy()
        end
    end

    for _, d in ipairs(clone:GetDescendants()) do
        if d:IsA("BasePart") then
            d.CanCollide = false
            d.Massless = true
            d.Anchored = false
        end
    end

    local hrp = clone:FindFirstChild("HumanoidRootPart")
    if not hrp then
        clone:Destroy()
        return nil
    end

    clone.PrimaryPart = hrp

    local hum = clone:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.PlatformStand = true
        hum.AutoRotate = false
    end

    local srcHRP = character:FindFirstChild("HumanoidRootPart")
    if srcHRP then
        clone:PivotTo(srcHRP.CFrame)
    end

    exploitState.invisible.mirrorModel = clone
    return hrp
end

local function updateInvisibleMirrorPosition(dt)
    local character = LocalPlayer.Character
    local invisibleState = exploitState.invisible
    if not character or not invisibleState.mirrorModel or not invisibleState.mirrorModel.PrimaryPart then
        return
    end

    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then
        return
    end

    local groundY = workspace.Map.Structural.Ground.Union.Position.Y
    local platformTopY = invisibleState.platform and (invisibleState.platform.Position.Y + invisibleState.platform.Size.Y * 0.5) or groundY

    local targetJumpHeight = math.max(0, (hrp.Position.Y - platformTopY) * 0.5)
    targetJumpHeight = math.min(targetJumpHeight, 20)

    local smoothing = math.clamp((dt or 1 / 60) * 10, 0, 1)
    invisibleState.lastJumpHeight = invisibleState.lastJumpHeight + (targetJumpHeight - invisibleState.lastJumpHeight) * smoothing

    local newPos = Vector3.new(hrp.Position.X, groundY + 3 + invisibleState.lastJumpHeight, hrp.Position.Z)
    local look = hrp.CFrame.LookVector
    local flatLook = Vector3.new(look.X, 0, look.Z)

    if flatLook.Magnitude > 0 then
        invisibleState.mirrorModel:PivotTo(CFrame.new(newPos, newPos + flatLook))
    else
        invisibleState.mirrorModel:PivotTo(CFrame.new(newPos))
    end
end

local function enableInvisible()
    local invisibleState = exploitState.invisible
    if invisibleState.active then
        return
    end

    local character = LocalPlayer.Character
    if not character then
        return
    end

    invisibleState.platform = createInvisiblePlatform()
    if not invisibleState.platform then
        return
    end

    invisibleState.mirrorPart = createInvisibleMirrorClone()
    if not invisibleState.mirrorPart then
        invisibleState.platform:Destroy()
        invisibleState.platform = nil
        return
    end

    invisibleState.originalCameraSubject = workspace.CurrentCamera.CameraSubject

    for _, p in ipairs(character:GetChildren()) do
        if p:IsA("BasePart") then
            p.CanCollide = false
        end
    end

    local hrp = character:FindFirstChild("HumanoidRootPart")
    local hum = character:FindFirstChildOfClass("Humanoid")
    if hrp and hum then
        local hip = hum.HipHeight
        local hrpHalf = hrp.Size.Y * 0.5
        local platformTopY = invisibleState.platform.Position.Y + invisibleState.platform.Size.Y * 0.5

        pcall(function()
            require(LocalPlayer.PlayerScripts.Character.FullCustomReplication).Override(character, CFrame.new(
                hrp.Position.X,
                platformTopY + hip + hrpHalf,
                hrp.Position.Z
            ))
        end)
    end

    local mirrorHum = invisibleState.mirrorModel:FindFirstChildOfClass("Humanoid")
    workspace.CurrentCamera.CameraSubject = mirrorHum or invisibleState.mirrorPart
    invisibleState.movementConnection = RunService.Heartbeat:Connect(updateInvisibleMirrorPosition)
    invisibleState.active = true
end

local function disableInvisible()
    local invisibleState = exploitState.invisible
    if not invisibleState.active then
        return
    end

    local character = LocalPlayer.Character
    if invisibleState.movementConnection then
        invisibleState.movementConnection:Disconnect()
        invisibleState.movementConnection = nil
    end

    if character and invisibleState.mirrorModel and invisibleState.mirrorModel.PrimaryPart then
        local hrp = character:FindFirstChild("HumanoidRootPart")
        local hum = character:FindFirstChildOfClass("Humanoid")
        if hrp and hum then
            for _, p in ipairs(character:GetChildren()) do
                if p:IsA("BasePart") then
                    p.CanCollide = true
                end
            end

            local hip = hum.HipHeight
            local hrpHalf = hrp.Size.Y * 0.5
            local groundY = workspace.Map.Structural.Ground.Union.Position.Y

            task.wait()
            pcall(function()
                require(LocalPlayer.PlayerScripts.Character.FullCustomReplication).Override(character, CFrame.new(
                    invisibleState.mirrorModel.PrimaryPart.Position.X,
                    groundY + hip + hrpHalf,
                    invisibleState.mirrorModel.PrimaryPart.Position.Z
                ))
            end)
            task.wait()

            workspace.CurrentCamera.CameraSubject = character:FindFirstChildOfClass("Humanoid") or hrp
        end
    elseif character then
        local hrp = character:FindFirstChild("HumanoidRootPart")
        for _, p in ipairs(character:GetChildren()) do
            if p:IsA("BasePart") then
                p.CanCollide = true
            end
        end
        workspace.CurrentCamera.CameraSubject = character:FindFirstChildOfClass("Humanoid") or hrp
    end

    if invisibleState.platform then
        invisibleState.platform:Destroy()
    end
    if invisibleState.mirrorModel then
        invisibleState.mirrorModel:Destroy()
    end

    invisibleState.platform = nil
    invisibleState.mirrorModel = nil
    invisibleState.mirrorPart = nil
    invisibleState.lastJumpHeight = 0
    invisibleState.active = false
end

    local alivePlayers = {}

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer
and player.Character then
    if Settings.KillFarmingIgnoreFriends and isFriendTarget(player) then
        continue
    end
            local humanoid = player.Character:FindFirstChild("Humanoid")
            local hrp = player.Character:FindFirstChild("HumanoidRootPart")
            local health = humanoid and (humanoid:GetAttribute("Health") or humanoid.Health or 0) or 0
            if humanoid and hrp and health > 0 and not humanoid:GetAttribute("Godmode") then

local function spectatePlayer(targetPlayer)
    local camera = workspace.CurrentCamera
    if not camera or not targetPlayer or not targetPlayer.Character then
        return
    end

    local humanoid = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        camera.CameraType = Enum.CameraType.Custom
        camera.CameraSubject = humanoid
        return
    end

    local hrp = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
    if hrp then
        camera.CameraType = Enum.CameraType.Scriptable
        camera.CFrame = CFrame.new(hrp.Position + Vector3.new(0, 10, 15), hrp.Position)
    end
end

local function findCharacterForColor(name)
    visualState.foundCharacters = {}
    for _, path in ipairs(colorPaths) do
        if path then
            local character = path:FindFirstChild(name)
            if character then
                table.insert(visualState.foundCharacters, character)
            end
        end
    end
    return #visualState.foundCharacters > 0
end

local function storeOriginalCharacterColors(obj)
    if not visualState.originalCharacterColors[obj] then
        visualState.originalCharacterColors[obj] = {}
    end
    for _, child in ipairs(obj:GetDescendants()) do
        if child:IsA("BasePart") then
            visualState.originalCharacterColors[obj][child] = { type = "BasePart", color = child.Color }
        elseif child:IsA("ParticleEmitter") then
            visualState.originalCharacterColors[obj][child] = { type = "ParticleEmitter", color = child.Color }
        elseif child:IsA("Trail") then
            visualState.originalCharacterColors[obj][child] = { type = "Trail", color = child.Color }
        elseif child:IsA("Beam") then
            visualState.originalCharacterColors[obj][child] = { type = "Beam", color = child.Color }
        elseif child:IsA("Highlight") then
            visualState.originalCharacterColors[obj][child] = { type = "Highlight", fillColor = child.FillColor, outlineColor = child.OutlineColor }
        elseif child:IsA("PointLight") then
            visualState.originalCharacterColors[obj][child] = { type = "PointLight", color = child.Color }
        end
    end
end

local function changeCharacterColors(obj, newColor)
    for _, child in ipairs(obj:GetDescendants()) do
        if child:IsA("BasePart") then
            child.Color = newColor
        elseif child:IsA("ParticleEmitter") or child:IsA("Trail") or child:IsA("Beam") then
            child.Color = ColorSequence.new(newColor)
        elseif child:IsA("Highlight") then
            child.FillColor = newColor
            child.OutlineColor = newColor
        elseif child:IsA("PointLight") then
            child.Color = newColor
        end
    end
end

local function restoreCharacterColors(obj)
    local saved = visualState.originalCharacterColors[obj]
    if not saved then
        return
    end
    for child, colorData in pairs(saved) do
        if child.Parent then
            if colorData.type == "Highlight" then
                child.FillColor = colorData.fillColor
                child.OutlineColor = colorData.outlineColor
            else

        end
    end
end

local function isHeadAccessory(acc)
    if not acc:IsA("Accessory") then
        return false
    end
    local handle = acc:FindFirstChild("Handle")
    if not handle then
        return false
    end
    local attachment = handle:FindFirstChildWhichIsA("Attachment")
    return attachment and (attachment.Name:lower():find("hair") or attachment.Name:lower():find("head"))
end

local function applyMaterialEffect(char, material, color)
    if not char then
        return
    end

    for _, obj in ipairs(char:GetChildren()) do
        if obj:IsA("Shirt") or obj:IsA("Pants") or obj:IsA("ShirtGraphic") then
            if not visualState.savedData.clothing[obj] then
                visualState.savedData.clothing[obj] = obj:Clone()
            end
            obj:Destroy()
        end
    end

    local head = char:FindFirstChild("Head")
    if head then
        local face = head:FindFirstChild("face")
        if face and not visualState.savedData.face then
            visualState.savedData.face = face:Clone()
            face:Destroy()
        end
    end

    for _, acc in ipairs(char:GetChildren()) do
        if isHeadAccessory(acc) then
            if not visualState.savedData.headAccessories[acc] then
                visualState.savedData.headAccessories[acc] = acc:Clone()
            end
            acc:Destroy()
        end
    end

    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            if not visualState.savedData.bodyParts[part] then
                visualState.savedData.bodyParts[part] = { material = part.Material, color = part.Color }
            end
            part.Material = material
            part.Color = color
        end
    end

    for _, acc in ipairs(char:GetChildren()) do
        if acc:IsA("Accessory") and acc:FindFirstChild("Handle") then
            local handle = acc.Handle
            for _, desc in ipairs(handle:GetDescendants()) do
                if desc:IsA("Decal") or desc:IsA("Texture") then
                    if not visualState.savedData.accessoryTextures[desc] then
                        visualState.savedData.accessoryTextures[desc] = desc.Texture
                    end
                    desc.Texture = ""
                elseif desc:IsA("SpecialMesh") then
                    if not visualState.savedData.meshTextures[desc] then
                        visualState.savedData.meshTextures[desc] = desc.TextureId
                    end
                    desc.TextureId = ""
                end
            end
            handle.Material = material
            handle.Color = color
        end
    end
end

local function revertMaterialEffect(char)
    if not char then
        return
    end

    for _, clone in pairs(visualState.savedData.clothing) do
        if clone and not char:FindFirstChild(clone.Name) then
            clone:Clone().Parent = char
        end
    end

    if visualState.savedData.face and char:FindFirstChild("Head") and not char.Head:FindFirstChild("face") then
        visualState.savedData.face:Clone().Parent = char.Head
    end

    for _, clone in pairs(visualState.savedData.headAccessories) do
        if clone and not char:FindFirstChild(clone.Name) then
            clone:Clone().Parent = char
        end
    end

    for inst, texture in pairs(visualState.savedData.accessoryTextures) do
        if inst and inst.Parent then
            inst.Texture = texture
        end
    end

    for mesh, textureId in pairs(visualState.savedData.meshTextures) do
        if mesh and mesh.Parent then
            mesh.TextureId = textureId
        end
    end

    for part, data in pairs(visualState.savedData.bodyParts) do
        if part and part.Parent then
            part.Material = data.material
            part.Color = data.color
        end
    end
end

local function reapplyMaterialEffect(char)
    if Settings.CharacterMaterial == "None" then
        revertMaterialEffect(char)
    else
        applyMaterialEffect(char, Enum.Material[Settings.CharacterMaterial], Settings.MaterialColor)
    end
end

local function processAuraObject(obj, mode, color)
    local function saveOriginalAuraColor(o)
        if o:IsA("ParticleEmitter") or o:IsA("Beam") or o:IsA("Trail") or o:IsA("PointLight") or o:IsA("SpotLight") or o:IsA("SurfaceLight") or o:IsA("MeshPart") or o:IsA("Part") then
            visualState.originalAuraColors[o] = o.Color
        elseif o:IsA("Highlight") then
            visualState.originalAuraColors[o] = { fillColor = o.FillColor, outlineColor = o.OutlineColor }
        end
    end

    local function applyAuraColor(o, c)
        if o:IsA("ParticleEmitter") or o:IsA("Beam") or o:IsA("Trail") then
            o.Color = ColorSequence.new(c)
        elseif o:IsA("PointLight") or o:IsA("SpotLight") or o:IsA("SurfaceLight") or o:IsA("MeshPart") or o:IsA("Part") then
            o.Color = c
        elseif o:IsA("Highlight") then
            o.FillColor = c
            o.OutlineColor = c
        end
    end

    local function resetAuraColor(o)
        local saved = visualState.originalAuraColors[o]
        if not saved then
            return
        end
        if o:IsA("Highlight") then
            o.FillColor = saved.fillColor
            o.OutlineColor = saved.outlineColor
        else
            o.Color = saved
        end
    end

    if mode == "save" then
        saveOriginalAuraColor(obj)
    elseif mode == "apply" and color then
        applyAuraColor(obj, color)
    elseif mode == "reset" then
        resetAuraColor(obj)
    end

    for _, child in pairs(obj:GetChildren()) do
        processAuraObject(child, mode, color)
    end
end

local function applyAura(auraName)
    Settings.AuraType = auraName
    clearAuras()

    local folder = visualState.auraPaths[auraName]
    if not folder then
        return
    end

    for _, effectPart in pairs(folder:GetChildren()) do
        local bodyPart = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild(effectPart.Name)
        if bodyPart then
            for _, effect in pairs(effectPart:GetChildren()) do
                local success, clone = pcall(function()
                    return effect:Clone()
                end)
                if success and clone then
                    clone.Parent = bodyPart
                    table.insert(visualState.activeAuras, clone)
                    processAuraObject(clone, "save")
                    if Settings.AuraColor and not Settings.RainbowAura then
                        processAuraObject(clone, "apply", Settings.AuraColor)
                    end
                end
            end
        end
    end
end

local function recolorAuras(color)
    for _, aura in pairs(visualState.activeAuras) do
        processAuraObject(aura, "apply", color)
    end
end

local function resetAurasToDefault()
    for _, aura in pairs(visualState.activeAuras) do
        processAuraObject(aura, "reset")
    end
end

local function rainbowCycle()
    if Settings.RainbowAura and #visualState.activeAuras > 0 then
        local t = tick() * 2
        recolorAuras(Color3.new((math.sin(t) + 1) / 2, (math.sin(t + 2) + 1) / 2, (math.sin(t + 4) + 1) / 2))
    end
end

local function getTrailColor()
    if Settings.RainbowTrail then
        return Color3.fromHSV(visualState.trailData.rainbowHue, 1, 1)
    end
    return Settings.TrailColor
end

local function updateTrailRainbow()
    if Settings.RainbowTrail then
        visualState.trailData.rainbowHue = (tick() * 0.1) % 1
    end
end

local function createTrailSegment(startPos, endPos)
    local distance = (endPos - startPos).Magnitude
    if distance < visualState.trailData.movementThreshold then
        return
    end

    local part = Instance.new("Part")
    part.Name = "TrailSegment"
    part.Size = Vector3.new(0.2, 0.2, distance)
    part.CFrame = CFrame.lookAt(startPos, endPos) * CFrame.new(0, 0, -distance / 2)
    part.Anchored = true
    part.CanCollide = false
    part.Color = getTrailColor()
    part.Material = Enum.Material.Neon
    part.Parent = workspace
    Debris:AddItem(part, 1.1)
    TweenService:Create(part, TweenInfo.new(1), { Transparency = 1 }):Play()
    return part
end

local function stopTrails()
    if visualState.trailData.connection then
        visualState.trailData.connection:Disconnect()
        visualState.trailData.connection = nil
    end
    for _, segment in ipairs(visualState.trailData.segments) do
        if segment and segment.Parent then
            segment:Destroy()
        end
    end
    visualState.trailData.segments = {}
end

local function startTrails()
    stopTrails()
    if not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        return
    end

    visualState.trailData.lastPosition = LocalPlayer.Character.HumanoidRootPart.Position
    visualState.trailData.connection = RunService.Heartbeat:Connect(function()
        if not Settings.EnableTrails or not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
            stopTrails()
            return
        end

        local hrp = LocalPlayer.Character.HumanoidRootPart
        local currentPosition = hrp.Position
        if (currentPosition - visualState.trailData.lastPosition).Magnitude > visualState.trailData.movementThreshold then
            updateTrailRainbow()
            createTrailSegment(visualState.trailData.lastPosition, currentPosition)
            visualState.trailData.lastPosition = currentPosition
        end
    end)
end

        newChar:WaitForChild("Head", 5)

            applyAura(Settings.AuraType)
        end

        if Settings.EnableTrails then
            startTrails()
        else
            stopTrails()
        end

        if Settings.Invisible then
            task.delay(0.2, function()
                enableInvisible()
            end)
        end
    end)
end

    else
        resetAurasToDefault()
    end
end

        running = false,

getgenv().ServerLaggerMethod3 = false

local function stopServerLaggerMethod3()
    local state = exploitState.serverLaggerMethod3
    if not state.running then return end
    
    state.running = false
    
    if state.connection then
        state.connection:Disconnect()
        state.connection = nil
    end
    
    if state.abilitySpam then
        pcall(function()
            state.abilitySpam:Stop()
        end)
        state.abilitySpam = nil
    end
    
    if state.originalCharacter and state.originalCharacter ~= "Mob" then
        pcall(function()
            ReplicatedStorage.Remotes.Character.ChangeCharacter:FireServer(state.originalCharacter)
        end)
    end
    
    state.originalCharacter = nil
end

local function startServerLaggerMethod3()
    local state = exploitState.serverLaggerMethod3
    if state.running then return end
    
    pcall(function()
        state.originalCharacter = LocalPlayer.Data.Character.Value
    end)
    
    local currentChar = LocalPlayer.Data.Character.Value
    if currentChar ~= "Mob" then
        pcall(function()
            ReplicatedStorage.Remotes.Character.ChangeCharacter:FireServer("Mob")
        end)
        task.wait(0.5)
    end
    
    local AbilitySpam = {
        enabled = false,
        connection = nil
    }
    
    function AbilitySpam:GetCurrentCharacter()
        local ok, res = pcall(function()
            return LocalPlayer.Data.Character.Value
        end)
        if ok and res then return res end
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        return hum and hum:GetAttribute("CharacterName") or "Unknown"
    end
    
    function AbilitySpam:HasAbility4(characterName)
        local ok, res = pcall(function()
            local chars = ReplicatedStorage:WaitForChild("Characters")
            local folder = chars:FindFirstChild(characterName)
            local ab = folder and folder:FindFirstChild("Abilities")
            return ab and ab:FindFirstChild("4") ~= nil
        end)
        return ok and res
    end
    
    function AbilitySpam:FindNearestPlayer()
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return nil end
        
        local nearest, dist = nil, math.huge
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                local tr = p.Character:FindFirstChild("HumanoidRootPart")
                local th = p.Character:FindFirstChild("Humanoid")
                if tr and th then
                    local hp = th:GetAttribute("Health")
                    if hp and hp > 0 then
                        local d = (hrp.Position - tr.Position).Magnitude
                        if d < dist then
                            dist = d
                            nearest = p
                        end
                    end
                end
            end
        end
        return nearest
    end
    
    function AbilitySpam:GetNearestPlayerCFrame()
        local p = self:FindNearestPlayer()
        return p and p.Character and p.Character.HumanoidRootPart and p.Character.HumanoidRootPart.CFrame or CFrame.new()
    end
    
    function AbilitySpam:UseAbility4()
        local charName = self:GetCurrentCharacter()
        if not self:HasAbility4(charName) then return end
        
        local target = self:FindNearestPlayer()
        if not target then return end
        
        local targetChar = target.Character
        local targetCF = self:GetNearestPlayerCFrame()
        
        pcall(function()
            local ability = ReplicatedStorage.Characters[charName].Abilities["4"]
            ReplicatedStorage.Remotes.Abilities.Ability:FireServer(ability, 9000000)
            
            local actions = {377, 380, 383, 384, 385, 387, 389}
            for i = 1, 7 do
                local args = {
                    ability,
                    charName .. ":Abilities:4",
                    i,
                    9000000,
                    {
                        HitboxCFrames = {targetCF, targetCF},
                        BestHitCharacter = targetChar,
                        HitCharacters = {targetChar},
                        Ignore = i > 2 and {ActionNumber1 = {targetChar}} or {},
                        DeathInfo = {},
                        BlockedCharacters = {},
                        HitInfo = {
                            IsFacing = not (i == 1 or i == 2),
                            IsInFront = i <= 2,
                            Blocked = i > 2 and false or nil
                        },
                        ServerTime = tick(),
                        Actions = i > 2 and {ActionNumber1 = {}} or {},
                        FromCFrame = targetCF
                    },
                    "Action" .. actions[i],
                    i == 2 and 0.1 or nil
                }
                
                if i == 7 then
                    args[5].RockCFrame = targetCF
                    args[5].Actions = {
                        ActionNumber1 = {
                            [target.Name] = {
                                StartCFrameStr = tostring(targetCF.X)..","..tostring(targetCF.Y)..","..tostring(targetCF.Z)..",0,0,0,0,0,0,0,0,0",
                                ImpulseVelocity = Vector3.new(1901,-25000,291),
                                AbilityName = "4",
                                RotVelocityStr = "0,0,0",
                                VelocityStr = "1.900635,0.010867,0.291061",
                                Duration = 2,
                                RotImpulseVelocity = Vector3.new(5868,-6649,-7414),
                                Seed = math.random(1,1e6),
                                LookVectorStr = "0.988493,0,0.151268"
                            }
                        }
                    }
                end
                
                ReplicatedStorage.Remotes.Combat.Action:FireServer(unpack(args))
            end
        end)
    end
    
    function AbilitySpam:Start()
        if self.connection then return end
        self.enabled = true
        self.connection = RunService.Heartbeat:Connect(function()
            if not self.enabled then return end
            self:UseAbility4()
            task.wait(0.5)
            if self.enabled then
                pcall(function()
                    local c = self:GetCurrentCharacter()
                    ReplicatedStorage.Remotes.Abilities.AbilityCanceled:FireServer(
                        ReplicatedStorage.Characters[c].Abilities["4"]
                    )
                end)
            end
            task.wait(0.001)
        end)
    end
    
    function AbilitySpam:Stop()
        if self.connection then
            self.connection:Disconnect()
            self.connection = nil
        end
        self.enabled = false
    end
    
    state.abilitySpam = AbilitySpam
    state.running = true
    
    AbilitySpam:Start()
    
    state.connection = RunService.Heartbeat:Connect(function()
        if not getgenv().ServerLaggerMethod3 or runtimeState.destroyed then
            stopServerLaggerMethod3()
        end
    end)
end

local function startAntiLagWallCombo()
    local state = exploitState.antiLagWallCombo
    if state.running then return end

    state.running = true

    local effectsFolder =
        workspace:FindFirstChild("Misc")
        and workspace.Misc:FindFirstChild("Effects")
        or workspace:FindFirstChild("Effects")

    if not effectsFolder then return end

    local function clean()
        pcall(function()
            for _, v in pairs(effectsFolder:GetChildren()) do
                local name = v.Name

                if name:match("Wall[%-_]?[Cc]ombo$")
                or name:match("[Ee]xplosion")
                or name:match("[Ss]moke")
                or name:match("[Ff]ire")
                or name:match("[Ss]parkles")
                or name:match("[Pp]article")
                or name:match("[Bb]last")
                or name:match("[Ss]lash")
                or name:match("[Ii]mpact")
                or name:match("[Dd]ebris")
                or name:match("[Vv]fx")
                or v:IsA("ParticleEmitter")
                or v:IsA("Trail")
                or v:IsA("Beam") then
                    pcall(function()
                        v:Destroy()
                    end)
                end
            end
        end)
    end

    local childAdded = effectsFolder.ChildAdded:Connect(function(child)
        if not state.running then return end

        task.spawn(function()
            task.wait(0.01)

            if not child or not child.Parent then return end

            local name = child.Name

            if name:match("Wall[%-_]?[Cc]ombo$")
            or name:match("[Ee]xplosion")
            or name:match("[Ss]moke")
            or name:match("[Ff]ire")
            or name:match("[Ss]parkles")
            or name:match("[Pp]article")
            or name:match("[Bb]last")
            or name:match("[Ss]lash")
            or name:match("[Ii]mpact")
            or name:match("[Dd]ebris")
            or name:match("[Vv]fx")
            or child:IsA("ParticleEmitter")
            or child:IsA("Trail")
            or child:IsA("Beam") then
                pcall(function()
                    child:Destroy()
                end)
            end
        end)
    end)

    table.insert(state.connections, childAdded)

    state.cleanupThread = task.spawn(function()
        while state.running do
            task.wait(0.5)
            clean()
        end
    end)

    task.spawn(clean)
end

local function startAntiLagAbility()
    local state = exploitState.antiLagAbility
    if state.running then return end

    state.running = true

    local function hookCameraShake()
        pcall(function()
            local Core = require(ReplicatedStorage:WaitForChild("Core"))
            local CameraShake = Core.Get("Camera","Shake")
            if CameraShake and type(CameraShake) == "table" then
                CameraShake.Shake = function(...)
                    return {StopSustain = function() end}
                end
            end
        end)
    end

    local function hookRocks()
        pcall(function()
            local Core = require(ReplicatedStorage:WaitForChild("Core"))
            local Rocks = Core.Get("Map","Rocks")
            if Rocks and type(Rocks) == "table" then
                Rocks.Constant = function() end
                Rocks.CreateRock = function() end
                Rocks.Launch = function() end
            end
        end)
    end

    local function hookDestruction()
        pcall(function()
            local Core = require(ReplicatedStorage:WaitForChild("Core"))
            local Destruction = Core.Get("Map","Destruction")
            if Destruction and type(Destruction) == "table" then
                Destruction.Process = function() end
                Destruction.Destroy = function() end
            end
        end)
    end

    local function hookVFX()
        pcall(function()
            local VFX = require(ReplicatedStorage.Assets.VFXHelp)
            if VFX and type(VFX) == "table" then
                VFX.CreateShowcaseVFX = function()
                    return {PrimaryPart = Instance.new("Part")}
                end
            end
        end)
    end

    hookCameraShake()
    hookRocks()
    hookDestruction()
    hookVFX()

    local KEYWORDS = {
        "MobAwakeningGrab","LandingSmall","LandingLarge","Fall","Leap","Beams",
        "BeamsSmoke","BeamsFire","DustBig","MobAirstrike","Projectile","Ring",
        "MobCutscene","MobUltRocks","RockShatter","MobWall","MobSeismic",
        "LeftBeam","RightBeam","WallHit","MobWalFinisher","Summon","Explosion",
        "Aura","Dust","Rocks","CCBlack","CCWhite","Smoke","Fire","Sparkles",
        "Particle","Blast","Slash","Impact","Debris","VFX","Glow","Beam",
        "Trail","Lightning","Earthquake","Shockwave","Wave","Burst","Flash",
        "Glare","Ripple","Splash","Mist","Fog","Cloud","Energy","Magic","Spell"
    }

    local function shouldDestroy(obj)
        if not obj or not obj.Name then return false end
        for _,k in ipairs(KEYWORDS) do
            if obj.Name:find(k) then
                return true
            end
        end
        return false
    end

    state.descendantConnection = workspace.DescendantAdded:Connect(function(obj)
        if not state.running then return end

        task.spawn(function()
            task.wait(0.01)

            if not obj or not obj.Parent then return end

            if obj:IsA("Model") or obj:IsA("BasePart") or obj:IsA("Beam")

                end
            end

    end)

        if not state.running then return end

            or obj.Name:find("CC") or obj.Name:find("ColorCorrection") then
                pcall(function()
                    obj:Destroy()
                end)
            end
        end)
    end)

    local cleanupThread = task.spawn(function()
        while state.running do
            task.wait(1)

            pcall(function()
                local effectsFolder = workspace:FindFirstChild("Misc") and workspace.Misc:FindFirstChild("Effects")
                if effectsFolder then
                    for _,v in pairs(effectsFolder:GetChildren()) do
                        if shouldDestroy(v) then
                            v:Destroy()
                        end
                    end
                end

                for _,v in pairs(workspace:GetChildren()) do
                    if v:IsA("Model") or v:IsA("Part") then
                        if shouldDestroy(v) then
                            v:Destroy()
                        end
                    end
                end
            end)
        end
    end)

    table.insert(state.connections,{Disconnect=function() pcall(task.cancel,cleanupThread) end})
end

    Callback = bindStateSetting("UiToggleKey", function(value)
        applyWindowKeybind(getUiToggleKeyEnum())
    end)
})

configNameBox = config:Textbox({
    Title = "Config Name",
    Placeholder = "default",
    Value = Settings.SelectedConfigName,
    ClearTextOnFocus = false,
    Callback = function(value)
        value = tostring(value or ""):gsub("^%s+", ""):gsub("%s+$", "")
        if value == "" then
            value = "default"
        end
        Settings.SelectedConfigName = value
        SaveManager.SetSelectedConfig(value)
        refreshConfigDropdown()
    end
})

configSelectorDropdown = config:Dropdown({
    Title = "Configs",
    List = SaveManager.ListConfigs(),
    Value = Settings.SelectedConfigName,
    Callback = function(value)
        value = tostring(value or ""):gsub("^%s+", ""):gsub("%s+$", "")
        if value == "" then
            value = "default"
        end
        Settings.SelectedConfigName = value
        SaveManager.SetSelectedConfig(value)
        if configNameBox and type(configNameBox.SetValue) == "function" then
            pcall(configNameBox.SetValue, configNameBox, value, true)
        end
    end
})

config:Button({
    Title = "Save Config",
    Callback = function()
        SaveManager.Save(Settings, Settings.SelectedConfigName)
        Settings.SelectedConfigName = SaveManager.GetSelectedConfig()
        refreshConfigDropdown()
        if configNameBox and type(configNameBox.SetValue) == "function" then
            pcall(configNameBox.SetValue, configNameBox, Settings.SelectedConfigName, true)
        end
    end
})

config:Button({
    Title = "Load Config",
    Callback = function()
        local ok = SaveManager.LoadInto(Settings, Settings.SelectedConfigName)
        if ok then
            if configNameBox and type(configNameBox.SetValue) == "function" then
                pcall(configNameBox.SetValue, configNameBox, Settings.SelectedConfigName, true)
            end
            if configSelectorDropdown and type(configSelectorDropdown.SetValue) == "function" then
                pcall(configSelectorDropdown.SetValue, configSelectorDropdown, Settings.SelectedConfigName, true)
            end
            syncUiFromSettings()
            applyRuntimeSettings()
        end
    end
})

config:Button({
    Title = "Delete Config",
    Callback = function()
        local ok, nextSelected = SaveManager.Delete(Settings.SelectedConfigName)
        if ok then
            Settings.SelectedConfigName = nextSelected or "default"
            refreshConfigDropdown()
            if configNameBox and type(configNameBox.SetValue) == "function" then
                pcall(configNameBox.SetValue, configNameBox, Settings.SelectedConfigName, true)
            end
        end
    end
})

config:Toggle({
    Flag = "AutoLoadConfig",
    Title = "Auto Load",
    Value = Settings.AutoLoadConfig,
    Callback = bindStateSetting("AutoLoadConfig", function(value)
        SaveManager.SetAutoLoad(value)
    end)
})

        end
        refreshConfigDropdown()
        if configNameBox and type(configNameBox.SetValue) == "function" then
            pcall(configNameBox.SetValue, configNameBox, Settings.SelectedConfigName, true)
        end
        syncUiFromSettings()
        applyRuntimeSettings()
    end)
end
